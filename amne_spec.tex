\section*{2. Telos Function -- Definition \& Properties}
\begin{align*}
    T(c,k,e) = \alpha c + \beta k - \gamma e
\end{align*}
with $c,k,e \in [0,1]$ normalized indicators:
\begin{itemize}
    \item $c$ = clarity $= f(\text{length}(\varphi(X)), \text{distinct values}, \ldots)$
    \item $k$ = consistency $= 1$ if $\neg\exists$ contradictory pair in active terms, else $0$ (or soft)
    \item $e$ = equivocation $= \max_{\text{pairs}} E(A,B)$ where $E(A,B) = \min(S(A),S(B))/\max(S(A),S(B))$
\end{itemize}

\subsection*{Theorem 1. Monotonicity (gradient direction)}
\begin{align*}
    \frac{\partial T}{\partial c} &= \alpha \geq 0\\
    \frac{\partial T}{\partial k} &= \beta \geq 0\\
    \frac{\partial T}{\partial e} &= -\gamma \leq 0
\end{align*}
\noindent $\Rightarrow T$ is monotonically non-decreasing in clarity \& consistency, monotonically non-increasing in equivocation.

\subsection*{Corollary 1.1}
Any local improvement in $c$ or $k$ (or reduction in $e$) weakly increases $T$.

\section*{3. Self-Refining Update Rule}
\begin{align*}
    \alpha_{t+1} &= \alpha_t + \eta (1 - T_t) c_t\\
    \beta_{t+1} &= \beta_t + \eta (1 - T_t) k_t\\
    \gamma_{t+1} &= \gamma_t - \eta (1 - T_t) e_t
\end{align*}
with learning rate $\eta > 0$ small.

\subsection*{Theorem 2. Local ascent guarantee}
Assume $c,k,e$ fixed in small neighborhood. Then $\Delta T = T_{t+1} - T_t \geq 0$ for sufficiently small $\eta$.

\subsection*{Proof sketch (first-order Taylor)}
\begin{align*}
    \Delta T &\approx \left(\frac{\partial T}{\partial \alpha}\right) \Delta \alpha + \left(\frac{\partial T}{\partial \beta}\right) \Delta \beta + \left(\frac{\partial T}{\partial \gamma}\right) \Delta \gamma\\
    &= c (1-T) \eta c + k (1-T) \eta k + e (1-T) \eta e\\
    &= \eta (1-T) (c^2 + k^2 + e^2) \geq 0
\end{align*}
Since $(1-T) \geq 0$ and squares are nonnegative, $\Delta T \geq 0$. Equality only when $T=1$ or $(c=k=e=0)$ -- degenerate case.
